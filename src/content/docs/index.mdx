import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid>
	<Card title="One API, many audiences">
		Every audience is a tenant with its own locales, venues, people and events. Scope a GraphQL request to one audience with its endpoint path.
	</Card>
	<Card title="Themed per audience">
		Each audience picks a design system preset and may override its accent and spectrum colors. The API checks every override for contrast.
	</Card>
	<Card title="Kept fresh by machines">
		Service tokens let workers such as the researcher verify venues and events and upsert them with provenance.
	</Card>
	<Card title="SDKs for both platforms">
		[AudienceKit](/sdks/audiencekit/) for an audience's own app and [StaffKit](/sdks/#staffkit) for staff tools, each for Swift and Kotlin, released together on one version.
	</Card>
</CardGrid>
